## Сервер для замеров

Качать из интернета в рецепте незачем и невежливо: мы бы сами стали тем, от кого страница учит защищаться. Поэтому сервер свой, на `ThreadingHTTPServer` из stdlib. Он отвечает не сразу, а через `DELAY`, и отдаёт `429` с `Retry-After`, если одновременно обрабатывается больше `LIMIT` запросов. Ещё у него три особых пути: `/flaky/` отвечает `503` на первый запрос к каждому адресу, `/missing/` всегда `404`, `/slow/` отвечает очень медленно. Сервер поднимается внутри блока и гаснет вместе с ним, ничего не остаётся висеть.

```python title="slow_server.py"
from __future__ import annotations

import threading
import time
from collections.abc import Iterator
from contextlib import contextmanager
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

DELAY = 0.05        # секунд на обычный ответ
LIMIT = 20          # одновременных соединений; больше -- 429
RETRY_AFTER = 1     # секунд в заголовке Retry-After
SLOW_DELAY = 3.0    # пути /slow/... отвечают очень медленно
BODY_BASE = 1000    # размер тела: BODY_BASE + номер из пути

PATH_OK = "/ok/"
PATH_FLAKY = "/flaky/"      # первый запрос к каждому пути -- 503
PATH_SLOW = "/slow/"
PATH_MISSING = "/missing/"  # всегда 404


class Stats:
    def __init__(self) -> None:
        self.lock = threading.Lock()
        self.active = 0
        self.peak = 0
        self.rejected = 0   # сколько ответов 429
        self.served = 0
        self.seen_flaky: set[str] = set()

    def reset(self) -> None:
        with self.lock:
            self.active = self.peak = self.rejected = self.served = 0
            self.seen_flaky.clear()


STATS = Stats()


class Handler(BaseHTTPRequestHandler):
    def do_GET(self) -> None:  # noqa: N802 -- имя задано http.server
        with STATS.lock:
            STATS.active += 1
            STATS.peak = max(STATS.peak, STATS.active)
            over_limit = STATS.active > LIMIT
            if over_limit:
                STATS.rejected += 1
        try:
            if over_limit:
                self._reply(429, b"", {"Retry-After": str(RETRY_AFTER)})
            elif self.path.startswith(PATH_MISSING):
                self._reply(404, b"not found")
            elif self.path.startswith(PATH_FLAKY) and self._first_time():
                self._reply(503, b"try later")
            else:
                time.sleep(SLOW_DELAY if self.path.startswith(PATH_SLOW) else DELAY)
                number = int(self.path.rsplit("/", 1)[1])
                self._reply(200, b"x" * (BODY_BASE + number))
        except (BrokenPipeError, ConnectionResetError):
            pass  # клиент ушёл, не дождавшись ответа
        finally:
            with STATS.lock:
                STATS.active -= 1
                STATS.served += 1

    def _first_time(self) -> bool:
        with STATS.lock:
            first = self.path not in STATS.seen_flaky
            STATS.seen_flaky.add(self.path)
            return first

    def _reply(self, status: int, body: bytes, headers: dict[str, str] | None = None) -> None:
        self.send_response(status)
        self.send_header("Content-Length", str(len(body)))
        for name, value in (headers or {}).items():
            self.send_header(name, value)
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, format: str, *args: object) -> None:  # noqa: A002
        pass  # тишина в выводе


class Server(ThreadingHTTPServer):
    daemon_threads = True        # потоки обработчиков не мешают выходу
    request_queue_size = 512     # очередь приёма: по умолчанию 5, сотня клиентов разом не влезет


@contextmanager
def running_server() -> Iterator[str]:
    """Поднимает сервер в потоке и отдаёт базовый адрес; на выходе останавливает."""
    STATS.reset()
    server = Server(("127.0.0.1", 0), Handler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    try:
        yield f"http://127.0.0.1:{server.server_port}"
    finally:
        server.shutdown()
        server.server_close()


if __name__ == "__main__":
    import urllib.request

    with running_server() as base:
        with urllib.request.urlopen(f"{base}{PATH_OK}5") as response:
            print(response.status, len(response.read()))
```

```text title="Вывод"
200 1005
```

Порт выбирает система (`0`), поэтому параллельные запуски не мешают друг другу. «Пик» в замерах ниже считает сам сервер: сколько обработчиков жило одновременно. Запрос, получивший 429, тоже занимает обработчик на миг, поэтому пик может слегка превысить `LIMIT`.

## Четыре подхода в одной таблице

Один набор: 120 адресов `/ok/N`. Четыре варианта без повторов и без обработки 429, чтобы увидеть сами подходы: последовательный цикл, `ThreadPoolExecutor`, `asyncio.gather` без лимита и `asyncio.gather` с `Semaphore`. В «без лимита» клиент явно снимает потолок (`max_connections=None`): по умолчанию у `httpx` стоит 100, и пример бы показал не тот эффект, о котором речь.

```python title="bench.py"
from __future__ import annotations

import asyncio
import time
from collections.abc import Callable
from concurrent.futures import ThreadPoolExecutor

import httpx

from slow_server import DELAY, LIMIT, PATH_OK, STATS, running_server

URL_COUNT = 120
THREADS = 16        # меньше LIMIT: сервер не получает больше, чем держит
ASYNC_PARALLEL = 16


def make_urls(base: str) -> list[str]:
    return [f"{base}{PATH_OK}{i}" for i in range(URL_COUNT)]


def sequential(urls: list[str]) -> int:
    with httpx.Client() as client:
        return sum(client.get(url).status_code == 200 for url in urls)


def threads(urls: list[str]) -> int:
    with httpx.Client() as client, ThreadPoolExecutor(THREADS) as pool:
        return sum(r.status_code == 200 for r in pool.map(client.get, urls))


async def async_unbounded(urls: list[str]) -> int:
    limits = httpx.Limits(max_connections=None)  # без ограничения клиента
    async with httpx.AsyncClient(limits=limits) as client:
        responses = await asyncio.gather(*(client.get(url) for url in urls))
    return sum(r.status_code == 200 for r in responses)


async def async_bounded(urls: list[str]) -> int:
    gate = asyncio.Semaphore(ASYNC_PARALLEL)

    async def one(client: httpx.AsyncClient, url: str) -> int:
        async with gate:
            return (await client.get(url)).status_code

    async with httpx.AsyncClient() as client:
        codes = await asyncio.gather(*(one(client, url) for url in urls))
    return sum(code == 200 for code in codes)


def measure(name: str, run: Callable[[list[str]], int]) -> None:
    with running_server() as base:
        started = time.perf_counter()
        ok = run(make_urls(base))
        elapsed = time.perf_counter() - started
    print(f"{name:<24} {elapsed:5.1f} c   200: {ok:>3}   429: {STATS.rejected:>3}   пик: {STATS.peak:>3}")


if __name__ == "__main__":
    print(f"URL: {URL_COUNT}, задержка {DELAY} c, лимит сервера {LIMIT}")
    measure("последовательно", sequential)
    measure("ThreadPoolExecutor", threads)
    measure("asyncio без лимита", lambda urls: asyncio.run(async_unbounded(urls)))
    measure("asyncio с лимитом", lambda urls: asyncio.run(async_bounded(urls)))
```

```text title="Вывод"
URL: 120, задержка 0.05 c, лимит сервера 20
последовательно            7.2 c   200: 120   429:   0   пик:   1
ThreadPoolExecutor         0.5 c   200: 120   429:   0   пик:  16
asyncio без лимита         0.1 c   200:  20   429: 100   пик:  25
asyncio с лимитом          0.6 c   200: 120   429:   0   пик:  16
```

| Вариант | Время | Успешных из 120 | Ответов 429 | Пик на сервере |
| --- | --- | --- | --- | --- |
| последовательно | 7 с | 120 | 0 | 1 |
| `ThreadPoolExecutor`, 16 потоков | 0,5 с | 120 | 0 | 16 |
| asyncio без лимита | 0,1 с | 20 | 100 | около 25 |
| asyncio, 16 одновременно | 0,6 с | 120 | 0 | 16 |

Время и пик в выводе от запуска к запуску слегка плавают (в таблице округлено). Условия: один прогон на ноутбуке автора, Python 3.15, 120 URL, задержка 0,05 с, лимит сервера 20, клиент и сервер в одном процессе. Смысл в соотношении, а не в секундах. Из таблицы три вывода:

- «Без лимита» самый быстрый, но только потому, что почти все запросы (100 из 120) получили отказ мгновенно. Скорость без результата ничего не стоит.
- Потоки и asyncio с лимитом выигрывают у цикла более чем в десять раз и не нарушают лимит сервера. Разница между ними мала: узкое место здесь задержка сервера, а не способ ждать. asyncio выигрывает, когда соединений нужно держать много: поток заметно тяжелее задачи.
- Раз 429 приходят, их надо уметь обрабатывать. Лимит параллелизма на клиенте защищает от них, но не заменяет повтор: чужой сервер может ответить 429 или 503 и при скромном потоке.

## Почему не задача на каждый URL

Самый короткий код: `gather(*(fetch(url) for url in urls))`. Он создаёт по задаче на URL, и все они живут одновременно. На сотне адресов этого не видно. На миллионе видно: у каждой задачи свой фрейм корутины, объект `Task` и `Future`. Ниже модель: миллион адресов, «запрос» это один `await asyncio.sleep(0)`, чтобы сеть не мешала. Меряем пик `tracemalloc` у двух схем: задача на URL и пул из 100 воркеров на очереди с `maxsize`. Список адресов в обоих случаях отдаёт генератор, чтобы он не попадал в счёт.

```python title="memory.py"
from __future__ import annotations

import asyncio
import tracemalloc
from collections.abc import Iterator

URL_COUNT = 1_000_000
WORKERS = 100
QUEUE_FACTOR = 2


def urls() -> Iterator[str]:
    for i in range(URL_COUNT):
        yield f"http://example.test/page/{i}"


async def fake_fetch(url: str) -> int:
    await asyncio.sleep(0)  # «сеть»: отдаём управление один раз
    return len(url)


async def task_per_url() -> int:
    tasks = [asyncio.create_task(fake_fetch(url)) for url in urls()]  # миллион живых задач
    return sum(await asyncio.gather(*tasks))


async def pool() -> int:
    queue: asyncio.Queue[str | None] = asyncio.Queue(maxsize=WORKERS * QUEUE_FACTOR)
    total = 0

    async def feed() -> None:
        for url in urls():
            await queue.put(url)
        for _ in range(WORKERS):
            await queue.put(None)

    async def work() -> None:
        nonlocal total
        while (url := await queue.get()) is not None:
            size = await fake_fetch(url)
            total += size  # не `total += await ...`: левая часть читается до await

    async with asyncio.TaskGroup() as group:
        group.create_task(feed())
        for _ in range(WORKERS):
            group.create_task(work())
    return total


def measure(name: str, coroutine_factory) -> None:  # noqa: ANN001 -- пример
    tracemalloc.start()
    total = asyncio.run(coroutine_factory())
    _, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    print(f"{name:<14} пик {peak / 1024 / 1024:7.1f} МиБ   (контроль: {total})")


if __name__ == "__main__":
    measure("задача на URL", task_per_url)
    measure("пул воркеров", pool)
```

```text title="Вывод"
задача на URL  пик  1252.2 МиБ   (контроль: 30888890)
пул воркеров   пик     0.2 МиБ   (контроль: 30888890)
```

Контрольная сумма совпала: оба варианта обработали один и тот же миллион. Пик у задачи на URL около 1,2 ГиБ (около 1,2 КиБ на задачу, и это только каркас без сетевых буферов), у пула 0,2 МиБ, и он не зависит от числа адресов. Условия: один прогон, Python 3.15, на 3.12 цифры такие же с точностью до процентов; `tracemalloc` замедляет прогон, но пик считает по аллокациям Python, а не по процессу. Про `+=` и `await` в одной строке: `total += await f()` читает `total` до ожидания, и параллельные воркеры теряют чужие прибавки; в блоке для этого лишняя строка. Устройство «воркеры вместо задач на элемент» разобрано в разделе «Ограничение параллелизма на 100 000 задач» заметки [[Таймауты и нагрузка в asyncio]].

## Рецепт download_all

Сигнатура: `download_all(urls, *, workers, deadline)` возвращает async-генератор `Result`. Устройство:

- Две очереди с `maxsize = workers * 2`. В первую фидер кладёт URL, из неё воркеры берут. Во вторую воркеры кладут готовые `Result`, из неё читает генератор. Потребитель медленный: вторая очередь заполняется, воркеры встают на `put`, первая заполняется, фидер встаёт и перестаёт читать итератор URL. Это и есть обратное давление, и ни один список не растёт.
- Клиент `httpx.AsyncClient` один на всё скачивание. `Limits(max_connections=workers)` не пускает больше соединений, чем воркеров. `Timeout` по фазам (`connect`, `read`, `write`, `pool`): общего «таймаута запроса» у `httpx` нет, `read` считается на каждое получение куска данных.
- Повтор только для `GET`, только на сетевые ошибки (`httpx.TransportError`: таймаут, обрыв, отказ соединения), `429` и `5xx`. `404` и прочие `4xx` не повторяются, `3xx` тоже: `httpx` по умолчанию не следует редиректам (`follow_redirects=False`), поэтому `3xx` приходит как есть, считается ошибкой (`error="HTTP 301"`) и на диск не пишется; на диск попадают только тела `2xx`. Если нужны редиректы, клиент создают с `follow_redirects=True` и `max_redirects` (в `download_all` такого параметра нет, правится строка с `httpx.AsyncClient`). Неверный адрес (`InvalidURL`, `UnsupportedProtocol`) тоже не повторяется. Пауза: экспонента `BACKOFF_BASE * 2 ** (попытка - 1)` с потолком `BACKOFF_CAP`, из неё берётся случайная часть (полный джиттер), чтобы воркеры не просыпались одновременно. Если сервер прислал `Retry-After`, ждём его (числом секунд или датой), но не дольше `RETRY_AFTER_CAP`, плюс небольшой разброс. Число попыток задаёт параметр `max_attempts`.
- Почему только `GET`. Повторённый `POST` может выполниться дважды: оплата списана, письмо отправлено, а ответ потерялся по пути. Идемпотентность метода это гарантия сервера, а не клиента; о повторе неидемпотентной операции см. раздел «Повтор неидемпотентной операции» страницы [[Починка декораторов]].
- `CancelledError` нигде не ловится. Все `except` ниже перечисляют конкретные исключения `httpx`, а широкие `except Exception` в `feed` и в воркере не задевают `CancelledError`: в современном Python это `BaseException`, и отмена проходит сквозь него. Ошибка внутри воркера (наш баг) пишется в лог со стеком и превращается в `Result` с ошибкой, воркер живёт.
- Запись на диск `await asyncio.to_thread(...)`, чтобы файловая операция не останавливала цикл (почему это важно, в разделе «Блокирующий код в async def» заметки [[Таймауты и нагрузка в asyncio]]).
- Дедлайн на всё скачивание: абсолютный момент `deadline_at` проверяется на каждом шаге. `asyncio.timeout_at` оборачивает ожидание очередной записи (так застрявший сервер не держит нас дольше срока), а после каждого `yield` генератор сверяет `loop.time()` с `deadline_at`. Одного таймаута вокруг ожидания мало: если потребитель медленнее воркеров, очередь `done` никогда не пуста, ожидание не длится и таймер не успевает сработать. Вокруг самого `yield` таймаут не ставят: он отменял бы чужой код потребителя. Цена выбора: дедлайн замечается, когда потребитель вернулся за следующим результатом, то есть с опозданием не больше одного шага потребителя. Если к моменту проверки остались только сигналы «воркер закончил» (все результаты уже отданы), опоздания нет и генератор выходит штатно. По дедлайну генератор бросает `TimeoutError`, в `finally` отменяет все свои задачи и закрывает клиент. Если нужен жёсткий предел с отменой потребителя посреди шага, весь `async with aclosing(...)` оборачивают в `asyncio.timeout(...)` снаружи.
- Финализация в `finally`: `cancel_and_wait` отменяет все задачи и ждёт каждую через `gather(return_exceptions=True)`, разбирая все результаты: отмена это норма, остальное идёт в лог.
- Фидер, упавший вместе с итератором URL, всё равно кладёт воркерам сигналы остановки (`None`), а исключение поднимается потребителю после того, как воркеры закончили. При отмене фидер сигналов не кладёт: воркеров отменяют следом.

`workers` и `max_attempts` меньше 1 дают `ValueError`: при `workers=0` очередь стала бы неограниченной.

Прогресс логируется раз в `progress_every` результатов. Фоновые задачи названы (`dl-worker-N`, `dl-feeder`): так их видно в `asyncio.all_tasks()`. Для отдельных фоновых задач вне пула там же готовый `spawn`/`shutdown`, здесь он не нужен: пул владеет своими задачами сам.

```python title="downloader.py"
from __future__ import annotations

import asyncio
import hashlib
import logging
import random
from collections.abc import AsyncIterator, Iterable
from dataclasses import dataclass
from datetime import datetime, timezone
from email.utils import parsedate_to_datetime
from pathlib import Path

import httpx

log = logging.getLogger("downloader")

DEFAULT_MAX_ATTEMPTS = 4
BACKOFF_BASE = 0.2            # секунд перед второй попыткой
BACKOFF_CAP = 5.0             # потолок экспоненты
RETRY_AFTER_CAP = 30.0        # не ждём дольше, даже если сервер просит
RETRY_AFTER_JITTER = 0.3      # разброс, чтобы воркеры не бросились одновременно
QUEUE_FACTOR = 2              # размер очереди = воркеры * QUEUE_FACTOR
DEFAULT_PROGRESS_EVERY = 50
DEFAULT_TIMEOUT = httpx.Timeout(connect=3.0, read=10.0, write=10.0, pool=5.0)

TOO_MANY_REQUESTS = 429
SERVER_ERRORS = range(500, 600)
SUCCESS_STATUSES = range(200, 300)  # 3xx -- не успех: httpx не следует редиректам сам, тела нет смысла сохранять


@dataclass(frozen=True)
class Result:
    url: str
    status: int | None
    size: int
    error: str | None
    attempts: int


def is_retryable_status(status: int) -> bool:
    return status == TOO_MANY_REQUESTS or status in SERVER_ERRORS


def parse_retry_after(value: str | None) -> float | None:
    """Retry-After бывает числом секунд или датой HTTP."""
    if value is None:
        return None
    try:
        return max(0.0, float(value))
    except ValueError:
        pass
    try:
        moment = parsedate_to_datetime(value)
    except (TypeError, ValueError):
        return None
    if moment.tzinfo is None:
        moment = moment.replace(tzinfo=timezone.utc)
    return max(0.0, (moment - datetime.now(timezone.utc)).total_seconds())


def retry_delay(attempt: int, retry_after: float | None) -> float:
    """Пауза после неудачной попытки номер attempt (с единицы)."""
    if retry_after is not None:
        return min(retry_after, RETRY_AFTER_CAP) + random.uniform(0, RETRY_AFTER_JITTER)
    backoff = min(BACKOFF_CAP, BACKOFF_BASE * 2 ** (attempt - 1))
    return random.uniform(0, backoff)  # «полный джиттер»


def _write(path: Path, body: bytes) -> None:
    path.write_bytes(body)


async def fetch(
    client: httpx.AsyncClient, url: str, *, max_attempts: int, save_dir: Path | None
) -> Result:
    """Один URL: только GET, повтор на сетевые ошибки, 429 и 5xx."""
    status: int | None = None
    error: str | None = None
    for attempt in range(1, max_attempts + 1):
        retry_after: float | None = None
        try:
            response = await client.get(url)
        except (httpx.UnsupportedProtocol, httpx.InvalidURL) as exc:
            return Result(url, None, 0, f"{type(exc).__name__}: {exc}", attempt)  # повтор не поможет
        except httpx.TransportError as exc:  # таймауты, обрыв, отказ в соединении
            status, error = None, type(exc).__name__
        except httpx.HTTPError as exc:  # остальное из httpx: повторять нечего
            return Result(url, None, 0, f"{type(exc).__name__}: {exc}", attempt)
        else:
            status = response.status_code
            if not is_retryable_status(status):
                body = response.content
                if save_dir is not None and status in SUCCESS_STATUSES:
                    name = hashlib.sha256(url.encode()).hexdigest()[:16]
                    try:
                        await asyncio.to_thread(_write, save_dir / name, body)
                    except OSError as exc:
                        return Result(url, status, len(body), f"OSError: {exc}", attempt)
                return Result(url, status, len(body), None if status in SUCCESS_STATUSES else f"HTTP {status}", attempt)
            error = f"HTTP {status}"
            retry_after = parse_retry_after(response.headers.get("Retry-After"))
        if attempt < max_attempts:
            await asyncio.sleep(retry_delay(attempt, retry_after))
    return Result(url, status, 0, error, max_attempts)


async def feed(urls: Iterable[str], queue: asyncio.Queue[str | None], workers: int) -> None:
    """Кладёт URL в ограниченную очередь; по концу -- по сигналу остановки на воркера."""
    failure: Exception | None = None
    try:
        for url in urls:
            await queue.put(url)  # очередь полна -- ждём, итератор не читается вперёд
    except Exception as exc:  # упал сам итератор URL: воркеров всё равно надо остановить
        failure = exc
    for _ in range(workers):
        await queue.put(None)
    if failure is not None:
        raise failure
    # CancelledError сюда не доходит: его никто не ловит, задача просто отменяется


async def worker(
    client: httpx.AsyncClient,
    todo: asyncio.Queue[str | None],
    done: asyncio.Queue[Result | None],
    *,
    max_attempts: int,
    save_dir: Path | None,
) -> None:
    while (url := await todo.get()) is not None:
        try:
            result = await fetch(client, url, max_attempts=max_attempts, save_dir=save_dir)
        except Exception as exc:  # баг в нашем коде: пишем в лог и в результат, воркер живёт
            log.exception("неожиданная ошибка на %s", url)
            result = Result(url, None, 0, f"{type(exc).__name__}: {exc}", 1)
        await done.put(result)  # потребитель медленный -- воркер ждёт здесь
    await done.put(None)  # «этот воркер закончил»


async def cancel_and_wait(tasks: list[asyncio.Task[None]]) -> None:
    for task in tasks:
        task.cancel()
    outcomes = await asyncio.gather(*tasks, return_exceptions=True)
    for outcome in outcomes:  # разбираем каждый: отмена -- норма, остальное в лог
        if isinstance(outcome, BaseException) and not isinstance(outcome, asyncio.CancelledError):
            log.error("задача завершилась с ошибкой", exc_info=outcome)


async def download_all(
    urls: Iterable[str],
    *,
    workers: int,
    deadline: float,
    max_attempts: int = DEFAULT_MAX_ATTEMPTS,
    timeout: httpx.Timeout = DEFAULT_TIMEOUT,
    save_dir: Path | None = None,
    progress_every: int = DEFAULT_PROGRESS_EVERY,
) -> AsyncIterator[Result]:
    """Отдаёт Result по мере готовности. По deadline секунд бросает TimeoutError.

    Закрывать только через contextlib.aclosing: тогда при раннем выходе
    воркеры отменяются, а клиент закрывается сразу, а не когда-нибудь.
    """
    if workers < 1 or max_attempts < 1:  # workers=0 сделал бы очередь без предела
        raise ValueError("workers и max_attempts должны быть не меньше 1")
    queue_size = workers * QUEUE_FACTOR
    todo: asyncio.Queue[str | None] = asyncio.Queue(maxsize=queue_size)
    done: asyncio.Queue[Result | None] = asyncio.Queue(maxsize=queue_size)
    limits = httpx.Limits(max_connections=workers, max_keepalive_connections=workers)
    loop = asyncio.get_running_loop()
    deadline_at = loop.time() + deadline
    async with httpx.AsyncClient(limits=limits, timeout=timeout) as client:
        feeder = asyncio.create_task(feed(urls, todo, workers), name="dl-feeder")
        tasks = [feeder] + [
            asyncio.create_task(
                worker(client, todo, done, max_attempts=max_attempts, save_dir=save_dir),
                name=f"dl-worker-{i}",
            )
            for i in range(workers)
        ]
        finished = count = 0
        try:
            while finished < workers:
                if loop.time() >= deadline_at:  # потребитель медленный: ожидание ниже не длится
                    # остались одни сигналы «воркер закончил» -- опоздания нет, иначе опоздали
                    if done.empty() or (item := done.get_nowait()) is not None:
                        raise TimeoutError("deadline")
                    finished += 1
                    continue
                async with asyncio.timeout_at(deadline_at):  # застрявшие воркеры; не вокруг yield
                    item = await done.get()
                if item is None:
                    finished += 1
                    continue
                count += 1
                if count % progress_every == 0:
                    log.info("готово %d", count)
                yield item
            await feeder  # если упал итератор URL, ошибка поднимется здесь
        finally:
            await cancel_and_wait(tasks)
```

Блок сам ничего не печатает, это модуль; демонстрации ниже его импортируют. Тот же пул переносится на `aiohttp`: меняется только `fetch` (сессия `aiohttp.ClientSession` с `TCPConnector(limit=workers)` и `ClientTimeout` вместо `Limits` и `Timeout`), очереди и воркеры остаются теми же.

Если потребитель бросил генератор, не закрыв его, `finally` отработает не сразу: только когда цикл финализирует генераторы (в `asyncio.run` это выход из `run`) или когда его соберёт сборщик мусора. До тех пор воркеры работают, а клиент держит соединения. Поэтому потребителю дан `aclosing`, и в докстринге это сказано прямо.

## Повторы и Retry-After

Два прогона. Первый на двух воркерах (меньше лимита сервера, 429 не будет): один адрес отвечает 503 на первый запрос и должен получиться со второй попытки, другой отвечает 404 и повторяться не должен. Второй: 60 адресов на сорока воркерах при лимите сервера 20. Сервер отвечает 429 с `Retry-After: 1`, воркеры ждут секунду с разбросом и добирают.

```python title="demo_retry.py"
from __future__ import annotations

import asyncio
from collections import Counter
from contextlib import aclosing

from downloader import Result, download_all
from slow_server import LIMIT, PATH_FLAKY, PATH_MISSING, PATH_OK, RETRY_AFTER, STATS, running_server

FEW_WORKERS = 2          # меньше LIMIT: 429 не будет
MANY_WORKERS = LIMIT * 2  # больше LIMIT: сервер начнёт отвечать 429
BURST = 60
DEADLINE = 60.0
MAX_ATTEMPTS = 4


async def collect(urls: list[str], workers: int) -> list[Result]:
    async with aclosing(
        download_all(urls, workers=workers, deadline=DEADLINE, max_attempts=MAX_ATTEMPTS)
    ) as results:
        return [result async for result in results]  # конечный набор известного размера


async def main() -> None:
    with running_server() as base:
        urls = [f"{base}{PATH_FLAKY}1", f"{base}{PATH_MISSING}1"]
        for result in sorted(await collect(urls, FEW_WORKERS), key=lambda r: r.url):
            path = result.url.removeprefix(base)
            print(f"{path:<10} status={result.status} attempts={result.attempts} error={result.error}")

        STATS.reset()
        burst = [f"{base}{PATH_OK}{i}" for i in range(BURST)]
        results = await collect(burst, MANY_WORKERS)
        by_attempts = Counter(result.attempts for result in results)
        print(f"{BURST} URL, воркеров {MANY_WORKERS}, лимит сервера {LIMIT}")
        print(f"все 200: {all(r.status == 200 for r in results)}")
        print(f"получили 429 хотя бы раз: {STATS.rejected > 0}, Retry-After: {RETRY_AFTER} c")
        print(f"URL с повтором: {sum(n for a, n in by_attempts.items() if a > 1) > 0}")


asyncio.run(main())
```

```text title="Вывод"
/flaky/1   status=200 attempts=2 error=None
/missing/1 status=404 attempts=1 error=HTTP 404
60 URL, воркеров 40, лимит сервера 20
все 200: True
получили 429 хотя бы раз: True, Retry-After: 1 c
URL с повтором: True
```

Здесь список собирается намеренно: `collect` берёт конечный набор известного размера. В реальном коде так не делают, это тема следующего раздела. Сколько именно 429 и повторов получилось, зависит от планировщика потоков, поэтому блок печатает только факты, которые не плавают.

Что считать причиной бана, а что нет. 429 с `Retry-After` это просьба подождать, и она исполняется. Бан приходит за игнорирование этой просьбы, за повторы без паузы и за параллелизм выше заявленного. Поэтому число воркеров должно быть не больше того, что сервис допускает: вежливые значения часто лежат в документации API.

## Результаты по мере готовности

Генератор отдаёт `Result` сразу, как только он готов, и потребитель решает, что с ним делать: записать в БД, отметить в чек-листе, посчитать статусы. Полный список не копится. Потребитель всегда оборачивает генератор в `contextlib.aclosing`: это гарантирует вызов `aclose()` и при `break`, и при исключении в теле цикла. Ниже сто адресов, десять воркеров, запись файлов на диск и прогресс раз в 25 результатов.

```python title="demo_basic.py"
from __future__ import annotations

import asyncio
import logging
import sys
import tempfile
from collections import Counter
from contextlib import aclosing
from pathlib import Path

from downloader import download_all
from slow_server import PATH_OK, STATS, running_server

URL_COUNT = 100
WORKERS = 10
DEADLINE = 30.0
PROGRESS_EVERY = 25


async def main() -> None:
    logging.basicConfig(level=logging.INFO, format="%(message)s", stream=sys.stdout)
    logging.getLogger("httpx").setLevel(logging.WARNING)  # httpx пишет строку на каждый запрос
    with running_server() as base, tempfile.TemporaryDirectory() as tmp:
        urls = (f"{base}{PATH_OK}{i}" for i in range(URL_COUNT))  # генератор, не список
        statuses: Counter[int | None] = Counter()
        total = 0
        async with aclosing(
            download_all(
                urls,
                workers=WORKERS,
                deadline=DEADLINE,
                save_dir=Path(tmp),
                progress_every=PROGRESS_EVERY,
            )
        ) as results:
            async for result in results:
                statuses[result.status] += 1
                total += result.size
        print(f"статусы: {dict(statuses)}, байт: {total}")
        print(f"файлов на диске: {len(list(Path(tmp).iterdir()))}, пик на сервере: {STATS.peak}")


asyncio.run(main())
```

```text title="Вывод"
готово 50
готово 75
готово 100
статусы: {200: 100}, байт: 104950
файлов на диске: 100, пик на сервере: 10
```

Пик на сервере равен числу воркеров: пул не выходит за свой потолок. Итоговый размер сходится с суммой `BODY_BASE + N` по сотне адресов.

## Ранний выход потребителя

Что будет, если потребителю хватило пяти результатов из десяти тысяч? Цикл делает `break`, `aclosing` вызывает `aclose()`, внутри генератора срабатывает `finally`: отменяются воркеры и фидер, закрывается клиент. Чтобы это было видно, блок на время демонстрации подменяет выход из клиента, печатая строку при закрытии, и считает живые задачи пула.

```python title="demo_break.py"
from __future__ import annotations

import asyncio
from contextlib import aclosing

import httpx

from downloader import download_all
from slow_server import PATH_OK, STATS, running_server

URL_COUNT = 10_000
WORKERS = 8
DEADLINE = 60.0
TAKE = 5

_original_aexit = httpx.AsyncClient.__aexit__


async def noisy_aexit(self: httpx.AsyncClient, *args: object) -> None:
    await _original_aexit(self, *args)  # type: ignore[arg-type]
    print("  httpx-клиент закрыт")


httpx.AsyncClient.__aexit__ = noisy_aexit  # type: ignore[method-assign]  # только для показа


def pending_downloader_tasks() -> list[str]:
    return sorted(t.get_name() for t in asyncio.all_tasks() if t.get_name().startswith("dl-"))


async def main() -> None:
    with running_server() as base:
        urls = (f"{base}{PATH_OK}{i}" for i in range(URL_COUNT))
        taken = 0
        async with aclosing(download_all(urls, workers=WORKERS, deadline=DEADLINE)) as results:
            async for result in results:
                taken += 1
                if taken == TAKE:
                    print(f"беру {TAKE} результатов и выхожу из цикла")
                    break
            print(f"задач пула во время выхода из цикла: {len(pending_downloader_tasks())}")
        print(f"после aclosing задач пула: {len(pending_downloader_tasks())}")
        print(f"сервер получил меньше 1% списка: {STATS.served < URL_COUNT // 100}")


asyncio.run(main())
```

```text title="Вывод"
беру 5 результатов и выхожу из цикла
задач пула во время выхода из цикла: 9
  httpx-клиент закрыт
после aclosing задач пула: 0
сервер получил меньше 1% списка: True
```

Внутри цикла задач пула девять (фидер и восемь воркеров), клиент ещё открыт. После выхода из `async with aclosing` их ноль и клиент закрыт. Остальные 9 990 адресов не тронуты: итератор URL не читался дальше очереди. Без `aclosing` цикл `break` оставил бы генератор незаконченным, и всё это зависело бы от сборщика мусора.

## Остановка по Ctrl+C

`asyncio.run` с 3.11 перехватывает `SIGINT`: отменяет главную задачу, даёт ей дойти до `finally`, потом поднимает `KeyboardInterrupt` (это описано в документации asyncio в разделе про `Runner`). Генератор в `async with aclosing` при отмене закрывается и отменяет свои задачи. Нажатие имитирует `signal.raise_signal(SIGINT)` через полсекунды; серверу даны трёхсекундные ответы, чтобы воркеры в этот момент были заняты.

```python title="demo_sigint.py"
from __future__ import annotations

import asyncio
import signal
from contextlib import aclosing

from downloader import download_all
from slow_server import PATH_SLOW, running_server

URL_COUNT = 1000
WORKERS = 4
DEADLINE = 60.0
SIGINT_AFTER = 0.5   # секунд до «нажатия Ctrl+C»


def pool_tasks() -> int:
    return sum(1 for t in asyncio.all_tasks() if t.get_name().startswith("dl-"))


async def main() -> None:
    asyncio.get_running_loop().call_later(SIGINT_AFTER, signal.raise_signal, signal.SIGINT)
    with running_server() as base:
        urls = (f"{base}{PATH_SLOW}{i}" for i in range(URL_COUNT))  # каждый ответ идёт 3 секунды
        try:
            async with aclosing(download_all(urls, workers=WORKERS, deadline=DEADLINE)) as results:
                async for _ in results:
                    pass
        finally:
            print(f"main прерван, задач пула осталось: {pool_tasks()}")


try:
    asyncio.run(main())
except KeyboardInterrupt:
    print("KeyboardInterrupt дошёл до верха, выход чистый")
```

```text title="Вывод"
main прерван, задач пула осталось: 0
KeyboardInterrupt дошёл до верха, выход чистый
```

В выводе нет «Task was destroyed but it is pending», и процесс завершился сам, без зависания и без ожидания трёх секунд на ответы. Так получилось потому, что воркеры отменяются и ждут отмену в `finally`, а `CancelledError` нигде не проглатывается. Задача, проглотившая отмену, этот прогон бы подвесила: `asyncio.run` ждал бы её бесконечно, об этом раздел «Проглоченная отмена» страницы [[Потерянные исключения в async]].

## Таймауты и дедлайн

Три защиты: таймаут на фазу ловит один зависший запрос, дедлайн ограничивает всё скачивание, а при медленном потребителе дедлайн проверяется после каждого результата. Первый прогон: сервер отвечает три секунды, чтение ограничено 0,5 с, попыток две; каждый адрес заканчивается `ReadTimeout` после двух попыток. Второй: таймаутов нет совсем (`httpx.Timeout(None)`), но дедлайн одна секунда, и генератор бросает `TimeoutError`, воркеры отменены. Третий: потребитель тратит 0,3 с на результат, дедлайн 0,5 с, адресов двадцать; без проверки после `yield` скачивание заняло бы около шести секунд и ошибки не было бы. Строка о закрытии клиента печатается подменой `__aexit__`, как в разделе «Ранний выход потребителя».

```python title="demo_timeouts.py"
from __future__ import annotations

import asyncio
import logging
import sys
import time
from contextlib import aclosing

import httpx

from downloader import download_all
from slow_server import PATH_OK, PATH_SLOW, running_server

READ_TIMEOUT = 0.5
SHORT_DEADLINE = 1.0
LONG_DEADLINE = 60.0
WORKERS = 4
ATTEMPTS = 2
SLOW_CONSUMER_PAUSE = 0.3
SLOW_URL_COUNT = 20
SLOW_DEADLINE = 0.5
MAX_SLOW = 1.5       # опоздание не больше одного шага потребителя
MAX_EXPECTED = 3.0   # воркеры вместо 3 секунд на ответ ждут от силы 1.5


_original_aexit = httpx.AsyncClient.__aexit__


async def noisy_aexit(self: httpx.AsyncClient, *args: object) -> None:
    await _original_aexit(self, *args)  # type: ignore[arg-type]
    print("  httpx-клиент закрыт")


httpx.AsyncClient.__aexit__ = noisy_aexit  # type: ignore[method-assign]  # только для показа


def pool_tasks() -> int:
    return sum(1 for t in asyncio.all_tasks() if t.get_name().startswith("dl-"))


async def main() -> None:
    logging.basicConfig(level=logging.INFO, format="%(message)s", stream=sys.stdout)
    logging.getLogger("httpx").setLevel(logging.WARNING)  # httpx пишет строку на каждый запрос
    with running_server() as base:
        urls = [f"{base}{PATH_SLOW}{i}" for i in range(WORKERS)]

        print("1) таймаут чтения 0.5 c, две попытки")
        started = time.perf_counter()
        timeout = httpx.Timeout(connect=1.0, read=READ_TIMEOUT, write=1.0, pool=1.0)
        async with aclosing(
            download_all(urls, workers=WORKERS, deadline=LONG_DEADLINE, max_attempts=ATTEMPTS, timeout=timeout)
        ) as results:
            async for result in results:
                print(f"  attempts={result.attempts} error={result.error}")
        print(f"  уложилось в {MAX_EXPECTED:.0f} c: {time.perf_counter() - started < MAX_EXPECTED}")

        print("2) таймаутов нет (read=None), но есть дедлайн 1 c")
        started = time.perf_counter()
        patient = httpx.Timeout(None)
        try:
            async with aclosing(
                download_all(urls, workers=WORKERS, deadline=SHORT_DEADLINE, timeout=patient)
            ) as results:
                async for result in results:
                    print(f"  пришёл {result.url}")
        except TimeoutError:
            print(f"  дедлайн сработал, уложились в {MAX_EXPECTED:.0f} c: {time.perf_counter() - started < MAX_EXPECTED}")

        print("3) быстрые адреса, но потребитель медленный (0.3 c на результат), дедлайн 0.5 c")
        fast = [f"{base}{PATH_OK}{i}" for i in range(SLOW_URL_COUNT)]
        started = time.perf_counter()
        taken = 0
        try:
            async with aclosing(
                download_all(fast, workers=WORKERS, deadline=SLOW_DEADLINE)
            ) as results:
                async for _ in results:
                    taken += 1
                    await asyncio.sleep(SLOW_CONSUMER_PAUSE)
        except TimeoutError:
            elapsed = time.perf_counter() - started
            print(f"  TimeoutError, уложились в {MAX_SLOW:.1f} c: {elapsed < MAX_SLOW}")
            print(f"  обработано {taken} из {SLOW_URL_COUNT}: {taken < SLOW_URL_COUNT}")
            print(f"  задач пула осталось: {pool_tasks()}")


asyncio.run(main())
```

Какое время сколько реально занимает, зависит от случайной паузы между попытками, поэтому блок печатает только «уложилось в 3 с». В третьем прогоне сервер быстрый, а потребитель медленный: адреса приходят быстрее, чем потребитель их забирает, и без проверки после `yield` дедлайн не сработал бы вовсе. Здесь он срабатывает с опозданием не больше одного шага потребителя (0,3 с). Все четыре адреса одинаковы, так что и строки одинаковы. Дедлайн не заменяет таймаут фазы: с таймаутом один зависший адрес занимает воркера на секунды, а без него до самого конца дедлайна, и остальные ждут. Таймаут по фазам описан в заметке [[Таймауты и нагрузка в asyncio]].

Две проверки границ: потребитель, медленный только на последнем результате, ошибку не получает (все данные отданы, остались сигналы остановки), а медленный с самого начала получает `TimeoutError`; `workers=0` отвергается сразу.

```python title="demo_deadline_last.py"
import asyncio
from contextlib import aclosing

from downloader import download_all
from slow_server import PATH_OK, running_server

PAUSE = 0.5
DEADLINE = 0.3


async def consume(urls: list[str], slow_from: int, workers: int) -> str:
    taken = 0
    try:
        async with aclosing(download_all(urls, workers=workers, deadline=DEADLINE)) as results:
            async for _ in results:
                taken += 1
                if taken >= slow_from:
                    await asyncio.sleep(PAUSE)
    except TimeoutError:
        return f"TimeoutError после {taken} из {len(urls)}"
    return f"без ошибки, получено {taken} из {len(urls)}"


async def main() -> None:
    with running_server() as base:
        urls = [f"{base}{PATH_OK}{i}" for i in range(6)]
        print("медленный с первого результата:", await consume(urls, 1, 2))
        print("медленный только на последнем:", await consume(urls, len(urls), 2))
        try:
            await consume([], 1, 0)
        except ValueError as exc:
            print("workers=0:", exc)


asyncio.run(main())
```

```text title="Вывод"
медленный с первого результата: TimeoutError после 1 из 6
медленный только на последнем: без ошибки, получено 6 из 6
workers=0: workers и max_attempts должны быть не меньше 1
```

## Как искать проблемы

**429 и бан.** Проверка: посчитать `Counter(r.status for r in results)` на прогоне; доля 429 выше нескольких процентов значит, что воркеров больше, чем допускает сервис, или `Retry-After` игнорируется. На стороне собственного сервиса смотрят в лог доступа по клиенту. Что сделать: снизить `workers` до заявленного лимита, проверить, что паузу берёт `retry_delay` (с `Retry-After`), а не фиксированная. Если сервис просит подождать минуты, а `RETRY_AFTER_CAP` меньше, воркеры будут стучаться раньше срока: потолок нужно поднять или остановить скачивание.

**Зависло на 99 %.** Проверка: новые результаты не приходят, а процесс живой. Нужно посмотреть, на чём стоят воркеры: `asyncio.all_tasks()` с именами `dl-worker-N`, а на уже запущенном процессе `py-spy dump --pid PID` (инструмент ставится `pip install py-spy`, вывод здесь не показан). Если воркеры висят внутри `client.get`, у запроса нет таймаута на какую-то фазу: проверить `Timeout`. Чаще всего зависает последний медленный адрес, пока остальные давно готовы. Что сделать: `Timeout` по всем четырём фазам, `max_attempts`, дедлайн на всё скачивание. Зависания потоков и `asyncio` в целом разобраны в разделе «Как искать зависания» страницы [[Выбор примитива синхронизации]].

**Память растёт.** Проверка: `tracemalloc` (см. «Почему не задача на каждый URL»), размеры очередей `queue.qsize()`, `len` любых списков результатов. Причины две: очередь без `maxsize` (фидер выкачал весь итератор вперёд) или список результатов вместо генератора. Что сделать: `maxsize` у обеих очередей, результат отдавать наружу, а не копить. Если нужна итоговая сводка, счётчики считают на лету, как `statuses` в «Результаты по мере готовности».

**Слишком много открытых файлов.** Проверка: `ulimit -n` в оболочке процесса, `len(os.listdir("/dev/fd"))` внутри (Linux и macOS) и `lsof -p PID | wc -l`. Каждое соединение это сокет, каждый открытый файл дескриптор, а у процесса есть потолок. В замерах выше «без лимита» клиент открывает сто и более сокетов, и на системе с `ulimit -n 256` это даёт `OSError: [Errno 24] Too many open files`, причём сервер в том же процессе делит дескрипторы с клиентом. Что сделать: `Limits(max_connections=workers)` (уже в рецепте), файлы открывать и закрывать коротко (`write_bytes` делает это сам, и через `to_thread` одновременно открыто не больше `workers` файлов), не поднимать `ulimit` вместо исправления.

## Что спросят дальше

- **Почему не `gather` по всем URL?** Он создаёт задачу на каждый адрес, и все они живут сразу: на миллионе это около гигабайта памяти и тысячи соединений разом. Пул из `workers` воркеров на очереди с `maxsize` держит в памяти только работу в полёте.
- **Какие запросы можно повторять?** Идемпотентные, то есть `GET` (и `HEAD`, `PUT`, `DELETE` по спецификации, но здесь не нужны), и только при сетевой ошибке, 429 или 5xx. `POST` повторяют только с ключом идемпотентности, который сервер действительно понимает.
- **Зачем джиттер в backoff?** Без него все воркеры, получившие ошибку в один момент, просыпаются в один момент и бьют сервер снова. Случайная часть паузы разводит их по времени.
- **Что делать с `Retry-After`?** Подождать столько, сколько просит сервер (число секунд или дата), но не дольше своего потолка, и добавить небольшой разброс.
- **Как не потерять `CancelledError`?** Не ловить `BaseException` и голый `except`, перечислять исключения, а широкий `except Exception` не задевает отмену. Если пришлось поймать отмену ради очистки, после неё делают `raise`.
- **Зачем `aclosing` у async-генератора?** Без него `break` не закрывает генератор: `finally` внутри ждёт сборщика мусора или финализатора цикла, и воркеры с клиентом висят. `aclosing` вызывает `aclose()` строго на выходе из блока.
- **Почему `asyncio.timeout` не обёрнут вокруг `yield`?** Отмена по таймауту прилетала бы в код потребителя посреди его шага. Генератор сам сверяет абсолютный дедлайн после каждого `yield` и бросает `TimeoutError`; жёсткий предел даёт `asyncio.timeout` снаружи вокруг всего `async with aclosing(...)`.
- **Чем ограниченная очередь помогает?** Она даёт обратное давление: медленный потребитель останавливает воркеров, воркеры останавливают фидера, итератор URL перестаёт читаться. Память не зависит от числа адресов.
- **Подойдёт ли это для `aiohttp`?** Да: очереди, воркеры и дедлайн те же, меняется только функция, которая делает запрос.
